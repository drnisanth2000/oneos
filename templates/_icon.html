{# Icon helpers over the sprite in _icon_sprite.html. Presentation only: every name maps to a symbol there, unknown values fall back to a folder. #} {% macro icon(name, cls="") -%} {%- endmacro %} {# A module's own name picks its icon (knowledge, legal, analytics, ... all sit in the "govern" block but mean different things). A key matches a whole hyphen-separated word of the module stem, singular or plural ("asset" matches "assets", never "reassets"); the first matching word wins, then the block icon, then a folder. #} {% set module_icons = [ ("inbox", "inbox"), ("archive", "archive"), ("foundation", "block-govern"), ("pipeline", "branch"), ("asset", "block-build"), ("operation", "wrench"), ("communication", "message"), ("people", "block-people"), ("finance", "block-finance"), ("sales", "block-growth"), ("marketing", "megaphone"), ("support", "lifebuoy"), ("knowledge", "book"), ("analytic", "chart"), ("legal", "scale"), ("self", "user"), ] %} {% macro block_icon(block, module="") -%} {%- set name = module.split("-", 1)[-1] if module[:1].isdigit() else module -%} {%- set found = namespace(icon="") -%} {%- for word in name.split("-") if not found.icon -%} {%- for key, symbol in module_icons if not found.icon and (word == key or word == key ~ "s") -%} {%- set found.icon = symbol -%} {%- endfor -%} {%- endfor -%} {%- if found.icon -%}{{ icon(found.icon) }} {%- elif block in ("system", "govern", "build", "deliver", "retain", "people", "finance", "growth") -%}{{ icon("block-" ~ block) }} {%- else -%}{{ icon("folder") }} {%- endif -%} {%- endmacro %} {% macro bundle_icon(flags) -%} {%- if "regulated" in flags -%}{{ icon("shield") }} {%- elif "personal" in flags -%}{{ icon("user") }} {%- elif "b2b" in flags -%}{{ icon("building") }} {%- elif "recurring" in flags -%}{{ icon("repeat") }} {%- else -%}{{ icon("folder") }} {%- endif -%} {%- endmacro %} {% macro file_icon(title) -%} {%- set lower = title | lower -%} {%- if lower.endswith(".eml") -%}{{ icon("mail") }} {%- elif lower.endswith((".jpg", ".jpeg", ".png", ".heic", ".gif", ".webp")) -%}{{ icon("image") }} {%- elif lower.endswith(".pdf") -%}{{ icon("file") }} {%- else -%}{{ icon("file-text") }} {%- endif -%} {%- endmacro %}
